cmake_minimum_required(VERSION 3.20)
project(pauli_path_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pauli_bench src/v1_circuits.cpp src/reference.cpp src/truncation.cpp src/monte_carlo.cpp src/terminal_aware_mc.cpp src/resampling_bfs.cpp src/hybrid_tail.cpp)
target_include_directories(pauli_bench PUBLIC include)
target_compile_options(pauli_bench PRIVATE -O3 -march=native -Wall -Wextra)
add_executable(pauli_benchmark benchmarks/main.cpp)
target_link_libraries(pauli_benchmark PRIVATE pauli_bench)
add_executable(pauli_mc_benchmark benchmarks/mc_main.cpp)
target_link_libraries(pauli_mc_benchmark PRIVATE pauli_bench)
add_executable(pauli_terminal_mc_benchmark benchmarks/terminal_mc_main.cpp)
target_link_libraries(pauli_terminal_mc_benchmark PRIVATE pauli_bench)
add_executable(pauli_resampling_benchmark benchmarks/resampling_bfs_main.cpp)
target_link_libraries(pauli_resampling_benchmark PRIVATE pauli_bench)
add_executable(pauli_hybrid_tail_one benchmarks/hybrid_tail_one.cpp)
target_link_libraries(pauli_hybrid_tail_one PRIVATE pauli_bench)
